\newcommand{\zjusecondpartchaptertocline}[2]
{
    \if\relax\detokenize{#2}\relax
        \par\noindent{\fangsong\zihao{-4}#1}\par
    \else
        \begingroup
            \renewcommand{\@dotsep}{\cftchapdotsep}%
            \renewcommand{\numberline}[1]{%
                \begingroup
                    \renewcommand{\hspace}[1]{}%
                    ##1%
                \endgroup
            }%
            \@dottedtocline{0}{0em}{0em}{\fangsong\zihao{-4}#1}{\fangsong\zihao{-4}#2}%
        \endgroup
    \fi
}
